    %% ===== 配色：冷色骨架承担数据通路，暖金标记门控，青绿标记实验支线 =====

    classDef modelBox  fill:#E9EFF6,stroke:#3D5A78,stroke-width:1.3px,color:#1B2A3F,rx:4px,ry:4px;
    classDef routerBox fill:#F8F1E2,stroke:#A98B3E,stroke-width:1.6px,color:#5C4617,rx:4px,ry:4px;
    classDef expBox    fill:#E8F2EF,stroke:#3F7D6E,stroke-width:1.3px,color:#1E463A,rx:4px,ry:4px;
    classDef mathBox   fill:#EDECF7,stroke:#5A56A8,stroke-width:1.3px,color:#2E2A66,rx:4px,ry:4px;
    classDef trainBox  fill:#EFF2F5,stroke:#2E3A48,stroke-width:1.3px,color:#141B24,rx:4px,ry:4px;
    classDef validBox  fill:#F6EBF1,stroke:#B2567B,stroke-width:1.3px,color:#5B2440,rx:4px,ry:4px,stroke-dasharray:4 3;

    %% ===== 顶部输入 =====
    Input["输入层<br/>模拟 (X,f,d,z) ｜ 实验 (X,f,d)"]:::inputBox

    %% ===== 第 1 带：双线并行结构 =====
    subgraph 双线并行结构 ["双线并行结构"]
        direction LR
        SimTrunk["物理主干网络<br/>(加载模型1权重)"]:::modelBox
        ExpFeature["特征提取器<br/>(h_exp)"]:::expBox
        ExpRes["残差网络<br/>(ΔP_exp Tanh 缩放)"]:::expBox

        subgraph 门控模块 ["小样本鲁棒门控"]
            direction LR
            BayesianGate["贝叶斯逆方差<br/>置信度估计"]:::routerBox
            Smoothing["方差平滑<br/>(Empirical Bayes)"]:::mathBox
            Clipping["截断保护<br/>(Clipping)"]:::mathBox
        end
    end

    %% ===== 第 2 带：参数融合与解析（分组框，仅为让它横排成带） =====
    subgraph 参数融合与解析 ["参数融合与解析"]
        direction LR
        AddParams["双线加性融合<br/>(物理尺度对齐)"]:::mathBox
        Alpha["缩放因子 α<br/>(Sigmoid 软门控 + 平滑约束)"]:::routerBox
        FinalAdd["最终参数计算<br/>τ_final = τ_汇聚 + α * Δτ"]:::mathBox
        SoftPhysProj["物理软约束投影<br/>(Softplus)"]:::mathBox
        Analytic["解析层<br/>(计算预测 RL)"]:::mathBox
    end

    %% ===== 第 3 带：推演引擎 =====
    subgraph 推演引擎 ["推演引擎"]
        direction LR
        TargetRL["目标 RL 输入<br/>(期望吸波性能)"]:::inputBox
        Forward["正向预测<br/>(冻结主干)"]:::modelBox
        Inverse["逆向设计<br/>(输出掺杂方案 X_inv)"]:::modelBox
        LoopCheck["循环一致性校验<br/>(独立正向评估)"]:::validBox
        Out1["输出预测结果"]:::modelBox
        Out2["输出帕累托最优方案"]:::modelBox
    end

    %% ===== 第 4 带：训练协议 =====

        direction LR
        LossSim["模拟 Loss<br/>(仅更新 Adapter)"]:::trainBox
        LossExp["实验 Loss<br/>(更新残差/门控参数/α)"]:::trainBox
        Scheduler["分阶段训练调度器"]:::trainBox
        SmoothConstraint["平滑约束<br/>(限制 α 变化率, 防灾难遗忘)"]:::trainBox
    end

    %% ===== 连线逻辑（与原图逐条一致） =====
    Input --> SimTrunk
    Input --> ExpFeature
    ExpFeature --> ExpRes
    ExpFeature --> BayesianGate
    BayesianGate --> Smoothing
    Smoothing --> Clipping

    TargetRL --> Inverse
    Inverse -->|输出 X_inv, z_inv| LoopCheck
    LoopCheck -.->|独立评估与筛选| Out2
    Inverse -.->|调用冻结主干验证| Forward
    Forward --> Out1

    SimTrunk --> AddParams
    ExpRes --> AddParams
    Clipping --> AddParams

    AddParams --> FinalAdd
    Alpha --> FinalAdd
    FinalAdd --> SoftPhysProj
    SoftPhysProj --> Analytic
    Analytic -->|全量预测 RL| Forward

    %% 训练虚线反馈
    Scheduler -.-> Alpha
    SmoothConstraint -.-> Alpha
    LossSim -.-> SimTrunk
    LossExp -.-> ExpRes
    LossExp -.-> Clipping
    LossExp -.-> Alpha
